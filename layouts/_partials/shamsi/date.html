$date.Year -}} {{- $breaks := slice -61 9 38 199 426 686 756 818 1111 1181 1210 1635 2060 2097 2192 2262 2324 2394 2456 3178 -}} {{- $year := 0 -}} {{- $found := false -}} {{- $dayOfYear := 0 -}} {{- /* Use civil dates at UTC midnight so daylight-saving changes cannot affect the day count. */ -}} {{- $civilDate := time.AsTime ($date.Format "2006-01-02") "UTC" -}} {{- range slice $gregorianYear (sub $gregorianYear 1) -}} {{- $candidate := sub . 621 -}} {{- if and (ge $candidate (index $breaks 0)) (lt $candidate 3178) -}} {{- $previous := index $breaks 0 -}} {{- $jalaliLeaps := -14 -}} {{- range after 1 $breaks -}} {{- $span := sub . $previous -}} {{- if lt $candidate . -}} {{- $elapsed := sub $candidate $previous -}} {{- $jalaliLeaps = add $jalaliLeaps (mul (div $elapsed 33) 8) (div (add (mod $elapsed 33) 3) 4) -}} {{- if and (eq (mod $span 33) 4) (eq (sub $span $elapsed) 4) -}} {{- $jalaliLeaps = add $jalaliLeaps 1 -}} {{- end -}} {{- break -}} {{- end -}} {{- $jalaliLeaps = add $jalaliLeaps (mul (div $span 33) 8) (div (mod $span 33) 4) -}} {{- $previous = . -}} {{- end -}} {{- $gregorianLeaps := sub (div . 4) (div (mul (add (div . 100) 1) 3) 4) 150 -}} {{- $marchDay := sub (add 20 $jalaliLeaps) $gregorianLeaps -}} {{- $nowruz := time.AsTime (printf "%04d-03-%02d" . $marchDay) "UTC" -}} {{- if not ($civilDate.Before $nowruz) -}} {{- $found = true -}} {{- $year = $candidate -}} {{- $dayOfYear = div (sub $civilDate.Unix $nowruz.Unix) 86400 -}} {{- break -}} {{- end -}} {{- end -}} {{- end -}} {{- if or (not $found) (ge $gregorianYear 3799) -}} {{- errorf "Cannot convert %s to Solar Hijri using the supported calendar range" ($date.Format "2006-01-02") -}} {{- end -}} {{- $month := add (div $dayOfYear 31) 1 -}} {{- $day := add (mod $dayOfYear 31) 1 -}} {{- if ge $dayOfYear 186 -}} {{- $remaining := sub $dayOfYear 186 -}} {{- $month = add (div $remaining 30) 7 -}} {{- $day = add (mod $remaining 30) 1 -}} {{- end -}} {{- $months := slice "فروردین" "اردیبهشت" "خرداد" "تیر" "مرداد" "شهریور" "مهر" "آبان" "آذر" "دی" "بهمن" "اسفند" -}} {{- return dict "year" $year "month" $month "monthName" (index $months (sub $month 1)) "day" $day -}}
